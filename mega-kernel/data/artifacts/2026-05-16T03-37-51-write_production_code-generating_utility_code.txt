function helloWorld(name) {
  try {
    if (typeof name !== 'string') {
      throw new Error('Name must be a string');
    }
    if (name.trim() === '') {
      throw new Error('Name cannot be empty');
    }
    console.log(Hello, ${name}!);
  } catch (error) {
    console.error(Error: ${error.message});
  }
}

helloWorld('World');
